# Everything installed from outside pacman/Flatpak, pinned. install.sh and
# update.sh build exactly these versions (scripts/lib/sources.sh); nothing
# floats to "whatever upstream has today". tools/bump-sources.sh moves pins
# to upstream's latest, VM-tests them, and keeps only the ones that pass.

#   git:     track = branch to follow, pin = commit
#   release: track = "latest" GitHub release, pin = tag
